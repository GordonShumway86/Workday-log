// Bump this string any time you deploy and want to force an update.
// It doesn't need to mean anything - just change it (v4, v5, v6...).
const CACHE_VERSION = 'v1';
const CACHE_NAME = 'workday-log-' + CACHE_VERSION;
const ASSETS = ['./', './index.html', './icon-180.png'];

self.addEventListener('install', (event) => {
  // Don't wait for old tabs to close - activate this version right away.
  self.skipWaiting();
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS))
  );
});

self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys()
      .then((names) => Promise.all(
        names.filter((n) => n !== CACHE_NAME).map((n) => caches.delete(n))
      ))
      .then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', (event) => {
  const req = event.request;

  // The app page itself: try the network first so you always get the
  // latest version when you have signal. Only fall back to the cached
  // copy if there's no connection.
  if (req.mode === 'navigate' || req.url.endsWith('index.html') || req.url.endsWith('/')) {
    event.respondWith(
      fetch(req)
        .then((res) => {
          caches.open(CACHE_NAME).then((cache) => cache.put(req, res.clone()));
          return res;
        })
        .catch(() => caches.match(req).then((r) => r || caches.match('./index.html')))
    );
    return;
  }

  // Everything else (icon, etc.): serve from cache if we have it,
  // otherwise fetch it.
  event.respondWith(
    caches.match(req).then((cached) => cached || fetch(req))
  );
});
